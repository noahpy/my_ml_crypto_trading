  8def create_feature_subplot(ax, data, title, color, label=None, second_data=None, second_color=None, second_label=None):
  9    """
 10    Create a single feature subplot with consistent styling and inside y-axis labels.
 11
 12    Args:
 13        ax: The matplotlib axis to plot on
 14        data: The data to plot
 15        title: The title for the subplot
 16        color: The color for the line

 19        second_color: Optional color for the second line
 20        second_label: Optional label for the second line (for legend)
 21    """
 22    # Plot the main data
 23    ax.plot(data, linewidth=1, color=color, label=label)
 24
 25    # Plot the optional second data if provided
 26    if second_data is not None and second_color is not None:
 27        ax.plot(second_data, linewidth=1,
 28                color=second_color, label=second_label)
 29
 30    # Add legend if we have labels
 31    if label is not None or second_label is not None:
 32        ax.legend(loc='upper right', fontsize=8)
 33
 34    ax.set_title(title, fontsize=10)
 35    ax.grid(True, alpha=0.3)
 36
 37    # Move the y-axis tick labels to the inside of the plot
 38    # First, hide the original labels
 39    ax.tick_params(axis='y', which='both', labelleft=False)

370    def create_feature(self, buffer: List[dict]) -> List[float]:
371        """
372        Create volatility feature from the buffer data.
373
374        Args:
375            buffer: List of order book and trade data
376
377        Returns:
378            List containing the volatility value
379        """
380        # Extract price data for each timestep in the buffer
381        prices = []
382
383        for snapshot in buffer[-self.timesteps_back:]:
384            # Get trades in this period
385            period_trades = snapshot["trades"]
386
387            if not period_trades:
388                # If no trades, use mid price from order book
389                mid_price = (max(snapshot["bids"].keys()) +
390                             min(snapshot["asks"].keys())) / 2
391                prices.append(mid_price)
392            else:
393                # Calculate VWAP for this period
394                vwap = sum(trade["price"] * trade["size"]
395                           for trade in period_trades)
396                total_vol = sum(trade["size"] for trade in period_trades)
397                if total_vol > 0:
398                    vwap = vwap / total_vol
399                else:
400                    vwap = (max(snapshot["bids"].keys()) +
401                            min(snapshot["asks"].keys())) / 2
402
403                prices.append(vwap)
404
405        # Calculate volatility as standard deviation of prices
406        volatility = np.std(prices) if len(prices) > 1 else 0
407
408        return [volatility]

375
376        Args:

378
379        Returns:
380            List containing the volatility value
381        """
382        # Extract price data for each timestep in the buffer
383        prices = []
384
385        for snapshot in buffer[-self.timesteps_back:]:
386            # Get trades in this period
387            period_trades = snapshot["trades"]
388
389            if not period_trades:
390                # If no trades, use mid price from order book
391                mid_price = (max(snapshot["bids"].keys()) +
392                             min(snapshot["asks"].keys())) / 2
393                prices.append(mid_price)
394            else:
395                # Calculate VWAP for this period
396                vwap = sum(trade["price"] * trade["size"]
397                           for trade in period_trades)
398                total_vol = sum(trade["size"] for trade in period_trades)
399                if total_vol > 0:
400                    vwap = vwap / total_vol
401                else:
402                    vwap = (max(snapshot["bids"].keys()) +
403                            min(snapshot["asks"].keys())) / 2
404
405                prices.append(vwap)
406
407        # Calculate volatility as standard deviation of prices
408        volatility = np.std(prices) if len(prices) > 1 else 0
409
410        return [volatility]
